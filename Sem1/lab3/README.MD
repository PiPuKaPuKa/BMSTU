### Условие задачи

Дан класс (например, с именем Vector), задающий вектор размерности n. Поля класса: указатель на массив, задающий вектор (тип элемента double), массив должен создаваться динамически, число элементов (размерность) вектора (тип int). Класс включает: конструктор без параметров, задающий пустой вектор (число элементов равно 0), конструктор, создающий объект вектор на основе обычного одномерного массива размерности n, конструктор копирования, конструктор перемещения, деструктор.

Необходимо перегрузить операции и продемонстрировать их работу. Перегрузить операцию [] (обращение к элементу вектора по индексу), операцию = (присваивание с копированием), операцию = (присваивание с перемещением), а также операцию вставки (<<) объекта в поток cout или в файл (объект класса ostream) и операцию извлечения (>>) объекта из потока cin или из файла (объект класса istream). Также продемонстрировать разницу между конструктором копирования и конструктором перемещения и между операциями присваивания с копированием и перемещением. Исходные коды класса разместить в двух файлах: в заголовочном файле класса и файле реализации класса.

При выполнении работы все входные данные читаются из текстового файла input.txt (создать этот файл любым текстовым редактором), результаты выводятся в файл output.txt. В отчете представить содержимое этих файлов.

### Текст программы

#### Vector.h

```cpp
#pragma once
#include <iostream>

// Класс "вектор" с динамическим массивом
class Vector {
public:
    Vector();                              // пустой вектор
    Vector(const double* arr, int size);   // вектор из обычного массива
    Vector(const Vector& other);           // конструктор копирования
    Vector(Vector&& other) noexcept;       // конструктор перемещения
    ~Vector();                             // деструктор

    Vector& operator=(const Vector& other);      // присваивание с копированием
    Vector& operator=(Vector&& other) noexcept;  // присваивание с перемещением

    double& operator[](int i);                   // обращение к элементу
    const double& operator[](int i) const;

    double operator*(const double* p) const;     // вариант 12: скалярное произведение

    int size() const { return n; }

    friend std::ostream& operator<<(std::ostream& os, const Vector& v);
    friend std::istream& operator>>(std::istream& is, Vector& v);

private:
    double* data;   // указатель на динамический массив
    int n;          // количество элементов
};
```

#### Vector.cpp

```cpp
#include "Vector.h"
#include <stdexcept>
#include <utility>

// Пустой вектор: памяти нет, размер 0
Vector::Vector() : data(nullptr), n(0) {}

// Вектор из обычного массива: выделяем память и копируем элементы
Vector::Vector(const double* arr, int size) : data(nullptr), n(size) {
    if (n > 0) {
        data = new double[n];
        for (int i = 0; i < n; ++i) data[i] = arr[i];
    }
}

// Копирование: создаем новую память и копируем данные.
// Исходный вектор не меняется.
Vector::Vector(const Vector& other) : data(nullptr), n(other.n) {
    if (n > 0) {
        data = new double[n];
        for (int i = 0; i < n; ++i) data[i] = other.data[i];
    }
}

// Перемещение: забираем чужую память, у исходного вектора ее не остается.
Vector::Vector(Vector&& other) noexcept : data(other.data), n(other.n) {
    other.data = nullptr;
    other.n = 0;
}

// Деструктор: освобождаем память
Vector::~Vector() {
    delete[] data;
}

// Присваивание с копированием
Vector& Vector::operator=(const Vector& other) {
    if (this != &other) {               // защита от a = a
        double* newData = nullptr;
        if (other.n > 0) {
            newData = new double[other.n];
            for (int i = 0; i < other.n; ++i) newData[i] = other.data[i];
        }
        delete[] data;                  // старую память удаляем только после создания новой
        data = newData;
        n = other.n;
    }
    return *this;
}

// Присваивание с перемещением
Vector& Vector::operator=(Vector&& other) noexcept {
    if (this != &other) {
        delete[] data;
        data = other.data;              // забираем память
        n = other.n;
        other.data = nullptr;           // у источника памяти больше нет
        other.n = 0;
    }
    return *this;
}

double& Vector::operator[](int i) {
    if (i < 0 || i >= n) throw std::out_of_range("индекс вне диапазона");
    return data[i];
}

const double& Vector::operator[](int i) const {
    if (i < 0 || i >= n) throw std::out_of_range("индекс вне диапазона");
    return data[i];
}

// Вариант 12: скалярное произведение вектора и обычного массива.
// Результат - одно число: сумма произведений соответствующих элементов.
double Vector::operator*(const double* p) const {
    if (n > 0 && p == nullptr) throw std::invalid_argument("нулевой указатель");

    double sum = 0.0;
    for (int i = 0; i < n; ++i) sum += data[i] * p[i];
    return sum;
}

// Вывод: сначала размер, потом элементы
std::ostream& operator<<(std::ostream& os, const Vector& v) {
    os << "Vector(" << v.n << "):";
    for (int i = 0; i < v.n; ++i) os << ' ' << v.data[i];
    return os;
}

// Ввод: сначала размер, потом элементы
std::istream& operator>>(std::istream& is, Vector& v) {
    int size = 0;
    if (!(is >> size) || size < 0) {
        is.setstate(std::ios::failbit);
        return is;
    }

    Vector tmp;
    if (size > 0) {
        tmp.data = new double[size];
        tmp.n = size;
        for (int i = 0; i < size; ++i) {
            if (!(is >> tmp.data[i])) return is;
        }
    }

    v = std::move(tmp);      // присваивание с перемещением
    return is;
}
```

    fin >> k;                       // размер обычного массива

    // 5. Конструктор копирования: копия не связана с исходником

    fout << "d = " << d << "\n";
    fout << "c после перемещения: " << c << "\n\n";

## Часть 2. Наследование и полиморфизм

Цель работы состоит в овладении навыками разработки программ на языке Си++, использующих возможности наследования классов для решения различных задач.

### Условие задачи (вариант 12)

Создать базовый класс «круг». Элементы класса: поле, содержащее значение радиуса круга (статус доступа protected); конструктор для инициализации поля; функция для печати радиуса круга. Создать производный класс «эллипс». Элементы класса: дополнительно поле, содержащее значение второй полуоси эллипса (для задания первой полуоси использовать наследуемое поле радиуса круга); конструктор для инициализации полей; переопределенная функция печати параметров эллипса (внутри переопределенной функции должна вызываться функция из базового класса). Создать по 1 объекту каждого из классов. Показать вызов созданных функций. При переопределении функций обеспечить и продемонстрировать два варианта: статический полиморфизм и динамический полиморфизм.

Создать вектор указателей на объекты классов, в одном векторе разные указатели на объекты базового и производного классов, напечатать данные всех объектов.

### Текст программы

```cpp
#include <iostream>
#include <vector>

#ifdef _WIN32
#include <Windows.h>
#endif

// Базовый класс: круг
class Circle {
protected:
    double radius;   // радиус доступен производному классу

public:
    // Конструктор: задает радиус
    Circle(double r) : radius(r) {}

    // Виртуальный деструктор нужен, когда объекты удаляются через указатель на базовый класс
    virtual ~Circle() {}

    // Статический полиморфизм: обычная (не виртуальная) функция.
    // Какая функция вызовется, определяет компилятор по ТИПУ указателя или объекта.
    void printStatic() const {
        std::cout << "Круг: радиус = " << radius << "\n";
    }

    // Динамический полиморфизм: виртуальная функция.
    // Какая функция вызовется, определяется во время работы по ТИПУ самого объекта.
    virtual void printDynamic() const {
        std::cout << "Круг: радиус = " << radius << "\n";
    }
};

// Производный класс: эллипс
class Ellipse : public Circle {
private:
    double secondAxis;   // вторая полуось

public:
    // Конструктор: первая полуось - это унаследованный радиус, вторая - отдельное поле
    Ellipse(double firstAxis, double second) : Circle(firstAxis), secondAxis(second) {}

    // Переопределение (статическое): скрывает функцию базового класса
    void printStatic() const {
        Circle::printStatic();                       // сначала функция базового класса
        std::cout << "   Вторая полуось = " << secondAxis << "\n";
    }

    // Переопределение (динамическое): override проверяет, что функция действительно переопределена
    void printDynamic() const override {
        Circle::printDynamic();                      // сначала функция базового класса
        std::cout << "   Вторая полуось = " << secondAxis << "\n";
    }
};

int main() {
#ifdef _WIN32
    SetConsoleOutputCP(CP_UTF8);
    SetConsoleCP(CP_UTF8);
#endif

    // 1. По одному объекту каждого класса
    Circle circle(3);
    Ellipse ellipse(5, 2);

    std::cout << "=== 1. Вызов функций у объектов ===\n";
    std::cout << "Объект Circle, printStatic:\n";
    circle.printStatic();
    std::cout << "Объект Ellipse, printStatic:\n";
    ellipse.printStatic();
    std::cout << "Объект Circle, printDynamic:\n";
    circle.printDynamic();
    std::cout << "Объект Ellipse, printDynamic:\n";
    ellipse.printDynamic();

    // 2. Указатель базового класса на объект производного класса
    Circle* ptr = &ellipse;

    std::cout << "\n=== 2. Статический полиморфизм (указатель Circle* на Ellipse) ===\n";
    ptr->printStatic();      // вызвана функция Circle: тип указателя определяет выбор

    std::cout << "\n=== 3. Динамический полиморфизм (указатель Circle* на Ellipse) ===\n";
    ptr->printDynamic();     // вызвана функция Ellipse: тип объекта определяет выбор

    // 4. Вектор указателей на объекты базового и производного классов
    std::cout << "\n=== 4. Вектор указателей ===\n";
    Circle* heapFigure = new Ellipse(7, 4);   // объект в динамической памяти

    std::vector<Circle*> figures;
    figures.push_back(&circle);      // базовый класс
    figures.push_back(&ellipse);     // производный класс
    figures.push_back(heapFigure);   // производный класс в динамической памяти

    for (std::size_t i = 0; i < figures.size(); ++i) {
        std::cout << "Объект " << i + 1 << ":\n";
        figures[i]->printDynamic();
    }

    delete heapFigure;   // освобождаем память, выделенную через new
    return 0;
}
```
